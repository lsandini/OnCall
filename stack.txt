version: "3.9"
services:
  oncall-frontend:
    image: "lsandini/oncall-frontend:latest"
    container_name: "oncall-frontend"
    restart: always
    networks:
      - "traefiknet"
    labels:
      traefik.enable: "true"
      traefik.http.routers.oncall-frontend.entrypoints: "websecure"
      traefik.http.routers.oncall-frontend.rule: "(Host(`oncall.cgmsim.com`)) && !PathPrefix(`/api`)"
      traefik.http.routers.oncall-frontend.tls.certresolver: "le"
      traefik.http.services.oncall-frontend.loadbalancer.server.port: 80
      traefik.http.routers.oncall-frontend.tls: true
      traefik.http.routers.oncall-frontend.middlewares: waf@docker

  oncall-backend:
    image: "lsandini/oncall-backend:latest"
    container_name: "oncall-backend"
    restart: always
    volumes:
      - oncall-data:/app/data
    environment:
      - NODE_ENV=production
      - PORT=3001
    networks:
      - "traefiknet"
    labels:
      traefik.enable: "true"
      traefik.http.routers.oncall-backend.entrypoints: "websecure"
      traefik.http.routers.oncall-backend.rule: "(Host(`oncall.cgmsim.com`)) && PathPrefix(`/api`)"
      traefik.http.routers.oncall-backend.tls.certresolver: "le"
      traefik.http.services.oncall-backend.loadbalancer.server.port: 3001
      traefik.http.routers.oncall-backend.tls: true
      traefik.http.routers.oncall-backend.middlewares: waf@docker

volumes:
  oncall-data:

networks:
  traefiknet:
    name: traefiknet
    attachable: true
